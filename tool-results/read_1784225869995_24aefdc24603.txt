   301→  function validateSignIn(): boolean {

   310→  function PasswordStrengthIndicator({ password }: { password: string }) {
   311→    const strength = getPasswordStrength(password);
   312→    if (!strength) return null;
   313→    return (
   314→      <div className="flex flex-col gap-1">
   315→        <div className="h-1 w-full rounded-full bg-[rgba(192,199,209,0.12)]">

   328→  function getPasswordStrength(pw: string): { label: string; color: string; width: string } | null {
   329→    if (!pw) return null;
   330→    let score = 0;
   331→    if (pw.length >= 8) score++;
   332→    if (pw.length >= 12) score++;
   333→    if (/[A-Z]/.test(pw)) score++;
   334→    if (/[a-z]/.test(pw)) score++;
   335→    if (/[0-9]/.test(pw)) score++;
   336→    if (/[^A-Za-z0-9]/.test(pw)) score++;
   337→    if (score <= 2) return { label: 'Weak', color: '#FF4757', width: '33%' };
   338→    if (score <= 4) return { label: 'Medium', color: '#FFD700', width: '66%' };
   339→    return { label: 'Strong', color: '#00E676', width: '100%' };
   340→  }
   341→
   342→  function validateRegister(): boolean {
   343→    const errs: Record<string, string> = {};
   344→    if (!regFirstName.trim()) errs.firstName = 'First name is required';
   345→    if (!regLastName.trim()) errs.lastName = 'Last name is required';
   346→    if (!regEmail.trim()) errs.email = 'Email is required';
   347→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(regEmail.trim())) errs.email = 'Invalid email address';
   348→    if (!regPhone.trim()) errs.phone = 'Phone number is required';
   349→    else if (!/^\d{7,15}$/.test(regPhone.replace(/\D/g, ''))) errs.phone = 'Phone number must be 7-15 digits';
   350→    if (!regPassword) errs.password = 'Password is required';
   351→    else if (regPassword.length < 8) errs.password = 'Password must be at least 8 characters';
   352→    else if (!/[A-Z]/.test(regPassword)) errs.password = 'Must contain at least 1 uppercase letter';
   353→    else if (!/[a-z]/.test(regPassword)) errs.password = 'Must contain at least 1 lowercase letter';
   354→    else if (!/[0-9]/.test(regPassword)) errs.password = 'Must contain at least 1 digit';
   355→    else if (regPassword === regEmail.trim()) errs.password = 'Password must not be the same as your email';
   356→    if (!regConfirm) errs.confirm = 'Please confirm your password';
   357→    else if (regPassword !== regConfirm) errs.confirm = 'Passwords do not match';
   358→    setFieldErrors(errs);
   359→    return Object.keys(errs).length === 0;
   360→  }

   369→        method: 'POST',
   370→        headers: { 'Content-Type': 'application/json' },

   372→      });

   375→        setError(data.error || 'Login failed');
   376→        return;
   377→      }
   378→      localStorage.setItem('brock_token', data.token);
   379→      localStorage.setItem('brock_user', JSON.stringify(data.user));
   380→      router.push('/');
   381→    } catch (_err) {
   382→      setError('Network error. Please check your connection.');

   384→      setLoading(false);
   385→    }
   386→  }
